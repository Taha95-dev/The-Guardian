File Example

#include <guardian/io/file.hpp>

using namespace guardian::io;

// Read a file
File file("data.txt");
if (file.is_open()) {
    std::string content = file.read_all();
    std::cout << content;
}

// Write a file
File output("output.txt");
output.write("Hello, Guardian!");

// Read lines
auto lines = file.read_lines();
for (const auto& line : lines) {
    std::cout << line << "\n";
}
